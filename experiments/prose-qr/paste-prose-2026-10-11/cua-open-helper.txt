async function openTaskWord(id){if(!/^word-0[4-9]$/.test(id))throw new Error('Task document id required');await word.pressKey('super+o');let ax=await word.getAXState({emit:false,disableDiffing:true});if(!ax.includes('open-panel'))throw new Error('Word Open panel required');await word.pressKey('super+shift+g');ax=await word.getAXState({emit:false,disableDiffing:true});let m=ax.match(/^\s*(\d+) text field \(settable\).*ID: PathTextField/m);if(!m)throw new Error('Go to field required');await word.setValue(Number(m[1]),'/Users/noir/Documents/aqrobat/docs/research/prose-qr/paste-prose-2026-10-11/'+id+'/'+id+'.docx');ax=await word.getAXState({emit:false,disableDiffing:true});if(!ax.includes(id+'.docx'))throw new Error('Exact task path required');await word.pressKey('Return');ax=await word.getAXState({emit:false,disableDiffing:true});m=ax.match(/^\s*(\d+) button Open, ID: OKButton/m);if(!m)throw new Error('Resolved input file required');await word.click(Number(m[1]));ax=await word.getAXState({emit:false,disableDiffing:true});if(!ax.includes('URL: file:///Users/noir/Documents/aqrobat/docs/research/prose-qr/paste-prose-2026-10-11/'+id+'/'+id+'.docx'))throw new Error('Exact task document not open');return{id,openedAt:new Date().toISOString()};}
