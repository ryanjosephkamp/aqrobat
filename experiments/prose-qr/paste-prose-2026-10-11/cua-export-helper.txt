async function exportTaskWordPdf(id){await word.pressKey('super+shift+s');let ax=await word.getAXState({emit:false});let m=ax.match(/^\s*(\d+) pop up button Word Document \(\.docx\)/m);if(!m)throw new Error('Expected Word Save As format control');await word.click(Number(m[1]));ax=await word.getAXState({emit:false});m=ax.match(/^\s*(\d+) PDF\s*$/m);if(!m)throw new Error('Expected PDF format item');await word.click(Number(m[1]));ax=await word.getAXState({emit:false});m=ax.match(/^\s*(\d+) radio button Best for printing, Value: ([01])/m);if(!m)throw new Error('Expected local printing export option');if(m[2]==='0'){await word.click(Number(m[1]));ax=await word.getAXState({emit:false});}m=ax.match(/^\s*(\d+) button Export, ID: OKButton/m);if(!m)throw new Error('Expected local PDF export');await word.click(Number(m[1]));ax=await word.getAXState({emit:false});if(!ax.includes(id+'.docx'))throw new Error('Expected task-owned document after export');return {id,route:'Word native Save As PDF, Best for printing',at:new Date().toISOString(),exportReturnedToTaskDocument:true};}
