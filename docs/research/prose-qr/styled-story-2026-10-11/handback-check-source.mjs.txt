const edits = [];
for (const name of editorial) {
  const old = await readFile(resolve(root, name), "utf8");
  assert.equal(sha(old), snapshot.files.find((f) => f.path === name).sha256);
  let text = old;
  for (const [a, b] of replacements) text = text.replaceAll(a, b);
  if (name === "README.md") {
    text = text.replace(
      "A [75-row ledger](index.html#ledger)",
      `A [75-row ledger](${resolve(root, "index.html")}#ledger)`,
    );
    text = text.replace(
      `[file hashes](${resolve(root, "evidence-sha256.json")})`,
      `[final file hashes](${resolve(root, "delivery-seal.json")})`,
    );
    text +=
      "\nThe initial evidence inventory remains a historical snapshot. Final handback copy edits, UI checks and current hashes are recorded separately in delivery-seal.json. The native images, HTML, text, frozen protocols, configurations and raw reader receipts were not edited.\n";
  } else {
    text = text.replace(
      "Heavier letters and more text per QR cell produce a stronger\n              overall pattern under ordinary reader settings.",
      "The size and font comparisons suggest that heavier letters and more text\n              per QR cell preserve the overall pattern under ordinary reader settings.",
    );
    text = text.replace(
      "grid-template-columns: 1.4fr 1fr;",
      "grid-template-columns: minmax(0, 1.4fr) minmax(0, 1fr);",
    );
    text = text.replace(
      '<a href="evidence-sha256.json">Evidence file hashes</a>',
      '<a href="delivery-seal.json">Final file hashes</a>',
    );
  }
  text = await format(text, {
    parser: name.endsWith("md") ? "markdown" : "html",
  });
  await writeFile(resolve(root, name), text);
  edits.push({
    path: name,
    beforeSha256: sha(old),
    afterSha256: sha(text),
    reason:
      "Handback prose spacing, qualified inference, current manifest link and responsive columns only",
  });
}

const snapshotFailures = [];

    (!edited && b.length !== f.bytes)

const files = [];
async function walk(dir) {
  for (const ent of (await readdir(dir, { withFileTypes: true })).sort((a, b) =>
    a.name.localeCompare(b.name),
  )) {
    const path = resolve(dir, ent.name);
    if (ent.isDirectory()) await walk(path);
    else {
      const b = await readFile(path);
      files.push({
        path: path.slice(cwd.length + 1),
        bytes: b.length,
        sha256: sha(b),
      });
    }
  }
}
await walk(root);
await walk(sourceRoot);
