import { chromium } from "playwright";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { resolve } from "node:path";
import { createHash } from "node:crypto";
import { spawnSync } from "node:child_process";
import { STORY } from "../flow/layout.mjs";
import { escapeHtml } from "../layout.mjs";
const root = resolve("docs/research/prose-qr/styled-story-2026-10-11/heavy-atlas-01");
await mkdir(root);
const specs = ["Impact", "Arial Black"].flatMap(font => [...new Set(STORY.toUpperCase())].sort().map(char => ({font,char})));
specs.forEach((s,i) => s.index=i);
const source = await readFile(new URL("heavy-atlas.mjs", import.meta.url));
const sha = b => createHash("sha256").update(b).digest("hex");
const save = (n,b) => writeFile(resolve(root,n), typeof b === "string" || Buffer.isBuffer(b) ? b : JSON.stringify(b,null,2)+"\n", {flag:"wx"});
await save("source.mjs.txt",source);
await save("manifest.json",{at:new Date().toISOString(),specs,sourceSha256:sha(source),size:20,lineHeight:20,captures:1,repeatSlots:1,readerCalls:0});
const html=`<!doctype html><html><body style="margin:0"><article id="artifact" style="position:relative;width:640px;height:${Math.ceil(specs.length/20)*32}px;background:white">${specs.map(s=>`<span id="g-${s.index}" style="position:absolute;left:${s.index%20*32}px;top:${Math.floor(s.index/20)*32}px;font:400 20px/20px '${s.font}';font-kerning:none;font-variant-ligatures:none;color:black">${escapeHtml(s.char)}</span>`).join("")}</article></body></html>`;
await save("atlas.html.txt",html);
const browser=await chromium.launch({headless:true,executablePath:"/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"});
try {
 const page=await browser.newPage({viewport:{width:800,height:800},deviceScaleFactor:1});
 await page.setContent(html); await page.evaluate(()=>document.fonts.ready);
 const cdp=await page.context().newCDPSession(page);await cdp.send("DOM.enable");await cdp.send("CSS.enable");const doc=await cdp.send("DOM.getDocument");
 const faces={};
 for(const font of ["Impact","Arial Black"]){const s=specs.find(s=>s.font===font&&s.char==="M");const node=await cdp.send("DOM.querySelector",{nodeId:doc.root.nodeId,selector:`#g-${s.index}`});faces[font]=(await cdp.send("CSS.getPlatformFontsForNode",{nodeId:node.nodeId})).fonts; if(!faces[font].some(f=>f.familyName===font))throw Error("Missing requested face "+font);}
 const advances=await page.evaluate(specs=>{const c=document.createElement("canvas").getContext("2d");c.fontKerning="none";return specs.map(s=>{c.font=`400 20px '${s.font}'`;const m=c.measureText(s.char);return {advance:m.width,ascent:m.actualBoundingBoxAscent,descent:m.actualBoundingBoxDescent};});},specs);
 const png=await page.locator("#artifact").screenshot();await save("atlas.png",png);const repeat=await page.locator("#artifact").screenshot();await save("repeat.png",repeat);if(!png.equals(repeat))throw Error("Repeat mismatch");
 const code="import cv2,sys,json,numpy as np\na=cv2.imread(sys.argv[1],0);out=[]\nfor i in range(int(sys.argv[2])):\n x=i%20*32;y=i//20*32;b=a[y:y+32,x:x+32];ys,xs=np.where(b<254);out.append({'inkMass':float(np.sum((255-b)/255)), 'inkBounds':[int(xs.min()),int(ys.min()),int(xs.max()-xs.min()+1),int(ys.max()-ys.min()+1)] if len(xs) else None})\nprint(json.dumps(out))";
 const raw=spawnSync("python3",["-B","-c",code,resolve(root,"atlas.png"),String(specs.length)],{encoding:"utf8"});await save("extract.stdout.txt",raw.stdout);await save("extract.stderr.txt",raw.stderr);if(raw.status!==0)throw Error("Measurement failed");
 const measure=JSON.parse(raw.stdout);const fonts={};
 specs.forEach((s,i)=>{(fonts[s.font]??={glyphs:{},faces:faces[s.font]}).glyphs[s.char]={...advances[i],...measure[i]};});
 for(const f of Object.values(fonts)){f.maxHeight=Math.max(...Object.values(f.glyphs).map(g=>g.inkBounds?.[3]||0));f.storyInkMass=[...STORY.toUpperCase()].reduce((sum,c)=>sum+f.glyphs[c].inkMass,0);f.storyAdvance=[...STORY.toUpperCase()].reduce((sum,c)=>sum+f.glyphs[c].advance,0);f.density=f.storyInkMass/(f.storyAdvance*18);}
 await save("results.json",{fonts,pngSha256:sha(png),repeatExact:true,browser:browser.version()});console.log(JSON.stringify(fonts));
} finally{await browser.close();}
