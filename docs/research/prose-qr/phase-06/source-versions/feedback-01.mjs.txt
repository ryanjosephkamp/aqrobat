async function bytes(p) {
  let sum = 0;
  for (const e of await readdir(p, { withFileTypes: true })) {
    const q = resolve(p, e.name);
    sum += e.isDirectory() ? await bytes(q) : (await stat(q)).size;
  }
  return sum;
}
async function save(p, b) {
  assert(
    (await bytes(root)) +
      (await bytes(resolve("experiments/prose-qr/finder-native"))) +
      Buffer.byteLength(b) <
      5_250_000,
    "Evidence reserve",
  );
  await writeFile(resolve(out, p), b, { flag: "wx" });
  return hash(b);
}

function score(d) {
  return Object.values(d.corridors)
    .flatMap((x) => Object.values(x))
    .reduce(
      (sum, a) =>
        sum +
        Math.round((1 - a.agreement) * a.pixels) +
        2 * Math.max(0, a.runs.length - 5),
      0,
    );
}
function nativeLayout(spec, lines) {
  const plainText = lines.join("\n"),
    tracking = seed.structural.tracking,
    actualAdvances = lines.map((s) =>
      [...s].reduce((n, c) => n + glyphs[c].advance + tracking, 0),
    );
  assert(
    actualAdvances.every((w) => w <= seed.field),
    "Native advance overflow",
  );
  const pad = seed.unit * 5,
    pre = (i, x, y) =>
      `<pre id="corner-${i}" style="position:absolute;left:${pad + x * seed.unit}px;top:${pad + y * seed.unit + seed.offset}px;width:${seed.field}px;height:${15 * seed.lineHeight}px;margin:0;padding:0;font:400 20px/${seed.lineHeight}px 'Impact',monospace;letter-spacing:${tracking}px;font-kerning:none;font-variant-ligatures:none;color:black;overflow:visible">${escapeHtml(plainText)}</pre>`;
  return {
    ...seed,
    spec,
    plainText,
    lines,
    actualAdvances,
    unusedWidths: actualAdvances.map((w) => seed.field - w),
    modelScores: [],
    matrix: finderMatrix(),
    markup: `<article id="artifact" style="position:relative;width:${seed.unit * 35}px;height:${seed.unit * 35}px;background:white;color:black;overflow:visible">${pre(0, 0, 0)}${pre(1, 18, 0)}${pre(2, 0, 18)}</article>`,
    structural: {
      ...seed.structural,
      quarterPixelRendererModel: false,
      actualNativePixelFeedback: true,
      wordBank: [...new Set(plainText.replaceAll(".", "").split(/\s+/))],
      semantic:
        "Complete and explicitly synthetic letter-words; sentence meaning not claimed",
    },
  };
}
